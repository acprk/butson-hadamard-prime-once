\section{Perfect quaternary sequences of length $N\equiv0\pmod4$}\label{app:quaternary}
% ===========================================================================

\begin{corollary}\label{cor:quaternary}
Let $N\equiv0\pmod4$, $N>16$. Exclude $N$ if one of the following
criteria applies: (C1) $N$ is not a sum of two squares
\cite[Thm.~4.5]{Arasu2011}; (C2) $N=2^a$, $a\ge5$
\cite[Thm.~3.6]{DucSchmidt2019}; (C3) an odd prime divisor of $N$ is
self-conjugate modulo $N$ \cite[Thm.~3.3]{DoDuc2019} (with $h=4$);
(C4) Schmidt's bound $N\le C^2F^2/(4\varphi(F))$, for a character of any
order $d\mid N$, $L=\operatorname{lcm}(4,d)$, $C=4N/L$ and
$F=F(L,N)$ \cite{Schmidt1999,LeungSchmidt2005,LeungSchmidt2012}, is
violated; (C5) Lemma~\ref{lem:Feq}; (C6) Theorem~\ref{thm:S}; (C7) for
$N\le1000$, \cite[Remark after Thm.~4.11]{Arasu2011} together with
Corollary~\ref{cor:arasu}. Then the lengths not excluded by these
criteria are \deptag{Comp}
\begin{itemize}
\item up to $5000$: $3528=2^3\cdot3^2\cdot7^2$ and $4900=2^2\cdot5^2\cdot7^2$;
\item up to $10^5$: in addition $7056$, $8712$, $9800$, $17424$, $19600$,
$34848$, $39200$, $44100$, $88200$ ($11$ in total);
\item up to $10^6$: in addition $108900$, $176400$, $213444$, $217800$,
$352800$, $426888$, $435600$, $544500$, $705600$, $736164$, $853776$,
$871200$ ($23$ in total).
\end{itemize}
\end{corollary}

We did not implement every lemma of \cite{ArasuDeLauneyMa2002}, so for
$N>1000$ these lists are upper bounds for the set of open lengths. Without
(C7), (C1)--(C6) leave in addition $72$, $100$, $144$, $784$. Up to
$10^5$, (C1)--(C5) exclude $19{,}382$ of the $25{,}000$ lengths,
Theorem~\ref{thm:S} excludes $5{,}604$ of the others, $3$ exist and $11$
remain. Table~\ref{tab:Feq} lists the lengths in $(1000,10^6]$ excluded
by (C5) alone; below $1000$, (C5) alone excludes $200$, $288$, $900$,
which are also excluded according to \cite{Arasu2011}. Odd $N>1$ are
impossible, and lengths $N\equiv2\pmod4$ with an odd prime of exponent
$1$ are excluded by Theorem~\ref{thm:S}; the other lengths
$N\equiv2\pmod4$ are not considered here.
